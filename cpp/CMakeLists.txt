cmake_minimum_required(VERSION 3.16)
project(ramanujan_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ramanujan_core src/ramanujan.cpp)
target_include_directories(ramanujan_core PUBLIC include)
add_executable(ramanujan_smoke src/main.cpp)
target_link_libraries(ramanujan_smoke PRIVATE ramanujan_core)
enable_testing()
add_test(NAME ramanujan_smoke COMMAND ramanujan_smoke)
